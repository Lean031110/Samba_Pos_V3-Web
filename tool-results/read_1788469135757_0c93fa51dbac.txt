    53→        public void UpdateEntity(Ticket ticket, Entity entity, int accountTypeId, int accountId, string entityCustomData)
    54→        {
    55→            var currentEntity = ticket.TicketEntities.SingleOrDefault(x => x.EntityTypeId == entity.EntityTypeId);
    56→            var currentEntityId = currentEntity != null ? currentEntity.EntityId : 0;
    57→            var newEntityName = entity.Name;
    58→            var oldEntityName = currentEntity != null ? currentEntity.EntityName : "";
    59→            var newEntityData = entityCustomData;
    60→            var oldEntityData = currentEntity != null ? currentEntity.EntityCustomData : "";
    61→
    62→            if (currentEntity != null && currentEntity.EntityId != entity.Id)
    63→            {
    64→                var entityType = _cacheService.GetEntityTypeById(currentEntity.EntityTypeId);
    65→                _applicationState.NotifyEvent(RuleEventNames.EntityUpdated, new
    66→                {
    67→                    Ticket = ticket,
    68→                    EntityTypeId = currentEntity.EntityTypeId,
    69→                    EntityId = currentEntity.EntityId,
    70→                    EntityTypeName = entityType.Name,
    71→                    OpenTicketCount = GetOpenTicketCount(currentEntity.EntityId, ticket.Id)
    72→                });
    73→            }
    74→
    75→            ticket.UpdateEntity(entity.EntityTypeId, entity.Id, entity.Name, accountTypeId, accountId, entityCustomData);
    76→
    77→            if (currentEntityId != entity.Id || oldEntityName != newEntityName || newEntityData != oldEntityData)
    78→            {
    79→                var entityType = _cacheService.GetEntityTypeById(entity.EntityTypeId);
    80→                _applicationState.NotifyEvent(RuleEventNames.TicketEntityChanged,
    81→                    new
    82→                    {
    83→                        Ticket = ticket,
    84→                        Entity = entity,
    85→                        EntityTypeId = entity.EntityTypeId,
    86→                        EntityId = entity.Id,
    87→                        EntityTypeName = entityType.Name,
    88→                        OldEntityName = oldEntityName,
    89→                        NewEntityName = newEntityName,
    90→                        OrderCount = ticket.Orders.Count,
    91→                        OldCustomData = oldEntityData,
    92→                        CustomData = newEntityData
    93→                    });
    94→            }
    95→        }

   139→        public TicketCommitResult CloseTicket(Ticket ticket)
   140→        {
   141→            var result = _ticketDao.CheckConcurrency(ticket);
   142→            Debug.Assert(ticket != null);
   143→            var changed = !string.IsNullOrEmpty(result.ErrorMessage);
   144→            var canSumbitTicket = !changed && ticket.CanSubmit;
   145→
   146→            if (canSumbitTicket)
   147→            {
   148→                RecalculateTicket(ticket);
   149→                _applicationState.NotifyEvent(RuleEventNames.BeforeTicketClosing, new { Ticket = ticket, TicketId = ticket.Id, ticket.RemainingAmount, ticket.TotalAmount });
   150→                if (ticket.Orders.Count > 0)
   151→                {
   152→                    var ticketType = _cacheService.GetTicketTypeById(ticket.TicketTypeId);
   153→
   154→                    if (ticket.Orders.Any(x => x.OrderNumber == 0))
   155→                    {
   156→                        var number = _settingService.GetNextNumber(ticketType.OrderNumerator.Id);
   157→                        ticket.MergeOrdersAndUpdateOrderNumbers(number);
   158→                    }
   159→
   160→                    if (ticket.Id == 0)
   161→                    {
   162→                        UpdateTicketNumber(ticket, ticketType.TicketNumerator);
   163→                        _ticketDao.Save(ticket);
   164→                    }
   165→
   166→                    Debug.Assert(!string.IsNullOrEmpty(ticket.TicketNumber));
   167→                    Debug.Assert(ticket.Id > 0);
   168→                    _applicationState.NotifyEvent(RuleEventNames.TicketClosing, new { Ticket = ticket, TicketId = ticket.Id, ticket.RemainingAmount, ticket.TotalAmount });
   169→                    ticket.LockTicket();
   170→                }
   171→
   172→                ticket.RemoveZeroAmountAccountTransactions();
   173→
   174→                if (ticket.Id > 0)// eğer adisyonda satır yoksa ID burada 0 olmalı.
   175→                    _ticketDao.Save(ticket);
   176→
   177→                Debug.Assert(ticket.Orders.Count(x => x.OrderNumber == 0) == 0);
   178→            }
   179→
   180→            if (ticket.Id > 0)
   181→            {
   182→                foreach (var ticketEntity in ticket.TicketEntities)
   183→                {
   184→                    var entityType = _cacheService.GetEntityTypeById(ticketEntity.EntityTypeId);
   185→                    _applicationState.NotifyEvent(RuleEventNames.EntityUpdated, new
   186→                    {
   187→                        EntityTypeId = ticketEntity.EntityTypeId,
   188→                        EntityId = ticketEntity.EntityId,
   189→                        EntityTypeName = entityType.Name,
   190→                        OpenTicketCount = _ticketServiceBase.GetOpenTicketIds(ticketEntity.EntityId).Count()
   191→                    });
   192→                }
   193→            }
   194→
   195→            result.TicketId = ticket.Id;
   196→            return result;
   197→        }

   199→        public void AddPayment(Ticket ticket, PaymentType paymentType, Account account, decimal amount, decimal tenderedAmount)
   200→        {
   201→            if (account == null) return;

   205→            _applicationState.NotifyEvent(RuleEventNames.PaymentProcessed,

   207→                {
   208→                    Ticket = ticket,
   209→                    PaymentTypeName = paymentType.Name,
   210→                    TenderedAmount = tenderedAmount,
   211→                    ProcessedAmount = tenderedAmount - changeAmount,
   212→                    ChangeAmount = changeAmount,
   213→                    SelectedQuantity = ticket.PaidItems.Sum(x => x.Quantity),
   214→                    RemainingAmount = ticket.GetRemainingAmount()
   215→                });
   216→        }

   254→        public TicketCommitResult MergeTickets(IEnumerable<int> ticketIds)
   255→        {
   256→            var ticketList = ticketIds.Select(OpenTicket).ToList();
   257→
   258→            if (ticketList.Any(x => x.Calculations.Any()))
   259→                return new TicketCommitResult { ErrorMessage = string.Format("Can't merge tickets\r{0}", "contains calculations") };
   260→
   261→            var entitiesUnMatches = ticketList.SelectMany(x => x.TicketEntities).GroupBy(x => x.EntityTypeId).Any(x => x.Select(y => y.EntityId).Distinct().Count() > 1);
   262→            if (entitiesUnMatches) return new TicketCommitResult { ErrorMessage = string.Format("Can't merge tickets\r{0}", "Entities doesn't match") };
   263→
   264→            var clonedOrders = ticketList.SelectMany(x => x.Orders).Select(ObjectCloner.Clone).ToList();
   265→            var clonedPayments = ticketList.SelectMany(x => x.Payments).Select(ObjectCloner.Clone2).ToList();
   266→            var clonedChangePayments = ticketList.SelectMany(x => x.ChangePayments).Select(ObjectCloner.Clone2).ToList();
   267→            var clonedTags = ticketList.SelectMany(x => x.GetTicketTagValues()).Select(ObjectCloner.Clone).ToList();
   268→            var clonedEntites = ticketList.SelectMany(x => x.TicketEntities).Select(ObjectCloner.Clone).ToList();
   269→            var clonedLogs = ticketList.SelectMany(x => x.GetTicketLogValues()).Select(ObjectCloner.Clone).ToList();
   270→            var ticketNumbers = string.Join(",", ticketList.Select(x => x.TicketNumber));
   271→
   272→            ticketList.ForEach(x => x.RemoveData());
   273→            ticketList.ForEach(x => CloseTicket(x));
   274→
   275→            var ticket = CreateTicket();
   276→
   277→            clonedOrders.ForEach(ticket.Orders.Add);
   278→
   279→            foreach (var cp in clonedPayments)
   280→            {
   281→                var account = _accountService.GetAccountById(cp.AccountTransaction.TargetTransactionValue.AccountId);
   282→                ticket.AddPayment(_cacheService.GetPaymentTypeById(cp.PaymentTypeId), account, cp.Amount, GetExchangeRate(account), cp.UserId);
   283→            }
   284→
   285→            foreach (var cp in clonedChangePayments)
   286→            {
   287→                var account = _accountService.GetAccountById(cp.AccountTransaction.TargetTransactionValue.AccountId);
   288→                ticket.AddChangePayment(_cacheService.GetChangePaymentTypeById(cp.ChangePaymentTypeId), account, cp.Amount, GetExchangeRate(account), cp.UserId);
   289→            }
   290→
   291→            clonedEntites.ForEach(x => ticket.UpdateEntity(x.EntityTypeId, x.EntityId, x.EntityName, x.AccountTypeId, x.AccountId, x.EntityCustomData));
   292→            clonedTags.ForEach(x => ticket.SetTagValue(x.TagName, x.TagValue));
   293→
   294→            ticket.SetLogs(clonedLogs);
   295→
   296→            RefreshAccountTransactions(ticket);
   297→
   298→            _applicationState.NotifyEvent(RuleEventNames.TicketsMerged, new { Ticket = ticket, TicketNumbers = ticketNumbers });
   299→            return CloseTicket(ticket);
   300→        }

   302→        public TicketCommitResult MoveOrders(Ticket ticket, Order[] selectedOrders, int targetTicketId)
   303→        {
   304→            _applicationState.NotifyEvent(RuleEventNames.TicketMoving, new { Ticket = ticket });
   305→            foreach (var selectedOrder in selectedOrders)
   306→            {
   307→                _applicationState.NotifyEvent(RuleEventNames.OrderMoving, new { Ticket = ticket, Order = selectedOrder, selectedOrder.MenuItemName, selectedOrder.Quantity });
   308→            }
   309→            var clonedOrders = selectedOrders.Select(ObjectCloner.Clone2).ToList();
   310→            ticket.RemoveOrders(selectedOrders);
   311→            CloseTicket(ticket);
   312→
   313→            var newTicket = OpenTicket(targetTicketId);
   314→
   315→            foreach (var clonedOrder in clonedOrders)
   316→            {
   317→                clonedOrder.TicketId = 0;
   318→                newTicket.Orders.Add(clonedOrder);
   319→                _applicationState.NotifyEvent(RuleEventNames.OrderMoved, new { Ticket = newTicket, Order = clonedOrder, clonedOrder.MenuItemName, clonedOrder.Quantity, OldTicketNumber = ticket.TicketNumber });
   320→            }
   321→
   322→            RefreshAccountTransactions(newTicket);
   323→            newTicket.LastOrderDate = DateTime.Now;
   324→
   325→            _applicationState.NotifyEvent(RuleEventNames.TicketMoved, new { Ticket = newTicket, OldTicketNumber = ticket.TicketNumber });
   326→
   327→            return CloseTicket(newTicket);
   328→        }

   330→        public void RecalculateTicket(Ticket ticket)
   331→        {
   332→            var total = ticket.TotalAmount;
   333→            ticket.Calculations.Where(x => x.CalculationType == 5).ToList().ForEach(
   334→                x => x.Amount = _expressionService.EvalCommand(FunctionNames.Calculation, "_" + x.Name, new { Ticket = ticket }, 0m));
   335→            ticket.Recalculate();
   336→            if (total != ticket.TotalAmount)
   337→            {
   338→                _applicationState.NotifyEvent(RuleEventNames.TicketTotalChanged,
   339→                    new
   340→                    {
   341→                        Ticket = ticket,
   342→                        PreviousTotal = total,
   343→                        TicketTotal = ticket.GetSum(),
   344→                        DiscountTotal = ticket.GetPreTaxServicesTotal(),
   345→                        PaymentTotal = ticket.GetPaymentAmount(),
   346→                        RemainingAmount = ticket.GetRemainingAmount()
   347→                    });
   348→            }
   349→        }

   423→        public void TagOrders(Ticket ticket, IEnumerable<Order> selectedOrders, OrderTagGroup orderTagGroup, OrderTag orderTag, string tagNote)
   424→        {
   425→            var so = selectedOrders.ToList();
   426→
   427→            if (orderTagGroup.MaxSelectedItems == 1)
   428→            {
   429→                foreach (var order in so.Where(x => x.OrderTagExists(y => y.OrderTagGroupId == orderTagGroup.Id && y.TagValue != orderTag.Name)))
   430→                {
   431→                    var orderTagValue = order.GetOrderTagValues().First(x => x.OrderTagGroupId == orderTagGroup.Id);
   432→                    order.UntagOrder(orderTagValue);
   433→                    _applicationState.NotifyEvent(RuleEventNames.OrderUntagged,
   434→                               new
   435→                               {
   436→                                   Ticket = ticket,
   437→                                   Order = order,
   438→                                   OrderTagName = orderTagGroup.Name,
   439→                                   OrderTagValue = orderTagValue.TagValue
   440→                               });
   441→
   442→                }
   443→            }
   444→
   445→            foreach (var selectedOrder in so)
   446→            {
   447→                var result = selectedOrder.ToggleOrderTag(orderTagGroup, orderTag, _applicationState.CurrentLoggedInUser.Id, tagNote);
   448→
   449→                if (orderTagGroup.SaveFreeTags && orderTagGroup.OrderTags.All(x => x.Name != orderTag.Name))
   450→                {
   451→                    _ticketDao.SaveFreeOrderTag(orderTagGroup.Id, orderTag);
   452→                    _cacheService.ResetOrderTagCache();
   453→                }
   454→                _applicationState.NotifyEvent(result ? RuleEventNames.OrderTagged : RuleEventNames.OrderUntagged,
   455→                new
   456→                {
   457→                    Ticket = ticket,
   458→                    Order = selectedOrder,
   459→                    OrderTagName = orderTagGroup.Name,
   460→                    OrderTagValue = orderTag.Name

   577→        public bool CanMakeAccountTransaction(TicketEntity ticketEntity, AccountTransactionType accountTransactionType, decimal targetBalance)
   578→        {
   579→            if (ticketEntity.AccountId == 0) return false;
   580→            var entityType = _cacheService.GetEntityTypeById(ticketEntity.EntityTypeId);
   581→            var typeId = accountTransactionType.TargetAccountTypeId;
   582→            if (accountTransactionType.DefaultSourceAccountId == 0)
   583→                typeId = accountTransactionType.SourceAccountTypeId;
   584→            var result = entityType.AccountTypeId == typeId;
   585→            if (result)
   586→            {
   587→                var accountType = _cacheService.GetAccountTypeById(entityType.AccountTypeId);
   588→                if (accountType.WorkingRule != 0)
   589→                {
   590→                    if (accountType.WorkingRule == 1 && targetBalance < 0) return false; //disallow credit
   591→                    if (accountType.WorkingRule == 2 && targetBalance > ticketEntity.GetCustomDataAsDecimal(Resources.CreditLimit)) return false; //disallow debit
   592→                }
   593→            }
   594→            return result;
   595→        }

   618→        public Order AddOrder(Ticket ticket, int menuItemId, decimal quantity, string portionName, string orderState)
   619→        {
   620→            if (ticket.IsLocked && !_userService.IsUserPermittedFor(PermissionNames.AddItemsToLockedTickets)) return null;
   621→            if (!ticket.CanSubmit) return null;
   622→            var menuItem = _cacheService.GetMenuItem(x => x.Id == menuItemId);
   623→            var portion = _cacheService.GetMenuItemPortion(menuItemId, portionName);
   624→            if (portion == null) return null;
   625→            var priceTag = _applicationState.CurrentDepartment.PriceTag;
   626→            var productTimer = _applicationState.GetProductTimer(menuItemId);
   627→
   628→            var order = ticket.AddOrder(
   629→                _applicationState.CurrentTicketType.SaleTransactionType,
   630→                _applicationState.CurrentDepartment.Model,
   631→                _applicationState.CurrentLoggedInUser.Name, menuItem,
   632→                _applicationState.GetTaxTemplates(menuItem.Id).ToList(), portion, priceTag, productTimer);
   633→
   634→            order.Quantity = quantity > 9 ? decimal.Round(quantity / portion.Multiplier, 3, MidpointRounding.AwayFromZero) : quantity;
   635→            order.ResetSelectedQuantity();
   636→            SetOrderState(order, orderState);
   637→
   638→            RecalculateTicket(ticket);
   639→            _applicationState.NotifyEvent(RuleEventNames.OrderAdded, new { Ticket = ticket, Order = order, MenuItemName = order.MenuItemName, MenuItemTag = menuItem.Tag, MenuItemGroupCode = menuItem.GroupCode });
   640→            return order;
   641→        }

   643→        private void SetOrderState(Order order, string orderState)
   644→        {
   645→            var i = 0;
   646→            var orderStates = orderState.Split(';');
   647→            foreach (var state in orderStates)
   648→            {
   649→                string gn;
   650→                string sv;
   651→                if (state.Contains("="))
   652→                {
   653→                    var sParts = state.Split('=');
   654→                    gn = sParts[0];
   655→                    sv = sParts[1];
   656→                }
   657→                else
   658→                {
   659→                    return;
   660→                }
   661→                order.SetStateValue(gn, 99 + i, sv, 99 + i, "", _applicationState.CurrentLoggedInUser.Id);
   662→                i++;
   663→            }
   664→        }
   665→    }
