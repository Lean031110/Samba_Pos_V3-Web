   354→   354→        private decimal CalculateServices(IEnumerable<Calculation> calculations, decimal sum)
   355→   355→        {
   356→   356→            decimal totalAmount = 0;
   357→   357→            var currentSum = sum;
   358→   358→
   359→   359→            foreach (var calculation in calculations.OrderBy(x => x.Order))
   360→   360→            {
   361→   361→                var sumValue = calculation.UsePlainSum ? Orders.Where(x => x.DecreaseInventory || x.IncreaseInventory).Sum(x => x.GetVisibleValue()) : sum;
   362→   362→
   363→   363→                calculation.Update(sumValue, currentSum, LocalSettings.Decimals);
   364→   364→
   365→   365→                totalAmount += calculation.CalculationAmount;
   366→   366→                currentSum += calculation.CalculationAmount;
   367→   367→
   368→   368→                if (calculation.Amount == 0 && calculation.CalculationType != 5)
   369→   369→                {
   370→   370→                    Calculations.Remove(calculation);
   371→   371→                }
   372→   372→
   373→   373→                calculation.UpdateCalculationTransaction(TransactionDocument, Math.Abs(calculation.CalculationAmount), ExchangeRate);
   374→   374→            }
   375→   375→
   376→   376→            return decimal.Round(totalAmount, LocalSettings.Decimals);
   377→   377→        }

   407→   407→            {

   656→   656→        public void Recalculate()
   657→   657→        {
   658→   658→            if (Orders.Count > 0)
   659→   659→            {
   660→   660→                var orderGroup = Orders.GroupBy(x => x.AccountTransactionTypeId);
   661→   661→                foreach (var orders in orderGroup)
   662→   662→                {
   663→   663→                    var o = orders;
   664→   664→                    var transaction = TransactionDocument.AccountTransactions.SingleOrDefault(x => x.AccountTransactionTypeId == o.Key);
   665→   665→                    if (transaction != null)
   666→   666→                    {
   667→   667→                        var amount = o.Sum(x => GetTaxExcludedSum(x));
   668→   668→                        transaction.UpdateAccounts(GetTicketAccounts());
   669→   669→                        transaction.UpdateAmount(amount, ExchangeRate);
   670→   670→                    }
   671→   671→                }
   672→   672→
   673→   673→                var taxIds = GetTaxIds();
   674→   674→                if (taxIds.Any())
   675→   675→                {
   676→   676→                    var plainSum = GetPlainSum();
   677→   677→                    var preTaxServices = GetPreTaxServicesTotal();
   678→   678→                    foreach (var taxId in taxIds)
   679→   679→                    {
   680→   680→                        var transaction = TransactionDocument.AccountTransactions.Single(x => x.AccountTransactionTypeId == taxId);
   681→   681→                        transaction.UpdateAccounts(GetTicketAccounts());
   682→   682→                        transaction.UpdateAmount(GetTaxTotal(taxId, preTaxServices, plainSum), ExchangeRate);
   683→   683→                    }
   684→   684→                }
   685→   685→            }
   686→   686→
   687→   687→            RemainingAmount = GetRemainingAmount();
   688→   688→            TotalAmount = GetSum();
   689→   689→        }

   734→   734→        private Order FixOrder(Order order)
   735→   735→        {
   736→   736→            Debug.Assert(order.SelectedQuantity > 0);
   737→   737→            Debug.Assert(Orders.Contains(order));
   738→   738→            if (order.SelectedQuantity >= order.Quantity) return null;
   739→   739→            var result = CloneOrder(order);
   740→   740→            result.Id = 0;
   741→   741→            result.Quantity = order.SelectedQuantity;
   742→   742→            result.ResetSelectedQuantity();
   743→   743→            order.Quantity -= order.SelectedQuantity;
   744→   744→            order.ResetSelectedQuantity();
   745→   745→            return result;
   746→   746→        }
