   308→        public decimal GetSum()
   309→        {
   310→            var plainSum = GetPlainSum();
   311→            var services = CalculateServices(Calculations.Where(x => !x.IncludeTax), plainSum);
   312→            var tax = TaxIncluded ? 0 : CalculateTax(plainSum, services);
   313→            plainSum = plainSum + services + tax;
   314→            services = CalculateServices(Calculations.Where(x => x.IncludeTax), plainSum);
   315→            return (plainSum + services);
   316→        }

   348→        public decimal CalculateTax(decimal plainSum, decimal preTaxServices)
   349→        {
   350→            var result = Orders.Where(x => x.CalculatePrice).Sum(x => x.GetTotalTaxAmount(TaxIncluded, plainSum, preTaxServices));
   351→            return decimal.Round(result, LocalSettings.Decimals);
   352→        }

   354→        private decimal CalculateServices(IEnumerable<Calculation> calculations, decimal sum)
   355→        {
   356→            decimal totalAmount = 0;
   357→            var currentSum = sum;
   358→
   359→            foreach (var calculation in calculations.OrderBy(x => x.Order))
   360→            {
   361→                var sumValue = calculation.UsePlainSum ? Orders.Where(x => x.DecreaseInventory || x.IncreaseInventory).Sum(x => x.GetVisibleValue()) : sum;
   362→
   363→                calculation.Update(sumValue, currentSum, LocalSettings.Decimals);
   364→
   365→                totalAmount += calculation.CalculationAmount;
   366→                currentSum += calculation.CalculationAmount;
   367→
   368→                if (calculation.Amount == 0 && calculation.CalculationType != 5)
   369→                {
   370→                    Calculations.Remove(calculation);
   371→                }
   372→
   373→                calculation.UpdateCalculationTransaction(TransactionDocument, Math.Abs(calculation.CalculationAmount), ExchangeRate);
   374→            }
   375→
   376→            return decimal.Round(totalAmount, LocalSettings.Decimals);
   377→        }

   379→        public void AddCalculation(CalculationType calculationType, decimal amount)
   380→        {
   381→            var calculation = Calculations.SingleOrDefault(x => x.CalculationTypeId == calculationType.Id) ??
   382→                    Calculations.SingleOrDefault(x => x.AccountTransactionTypeId == calculationType.AccountTransactionType.Id);
   383→            if (calculation == null)
   384→            {
   385→                calculation = new Calculation
   386→                        {
   387→                            Amount = amount,
   388→                            Name = calculationType.Name,
   389→                            CalculationType = calculationType.CalculationMethod,
   390→                            CalculationTypeId = calculationType.Id,
   391→                            IncludeTax = calculationType.IncludeTax,
   392→                            DecreaseAmount = calculationType.DecreaseAmount,
   393→                            UsePlainSum = calculationType.UsePlainSum,
   394→                            Order = calculationType.SortOrder,
   395→                            AccountTransactionTypeId = calculationType.AccountTransactionType.Id
   396→                        };
   397→                Calculations.Add(calculation);
   398→                TransactionDocument.AddSingletonTransaction(calculation.AccountTransactionTypeId, calculationType.AccountTransactionType, GetTicketAccounts());
   399→            }
   400→            else if (calculationType.ToggleCalculation && calculation.Amount == amount)
   401→            {
   402→                amount = 0;
   403→            }
   404→            else calculation.Amount = amount;
   405→            calculation.Name = calculationType.Name;
   406→            if (amount == 0 && calculation.CalculationType != 5)
   407→            {
   408→                Calculations.Remove(calculation);
   409→                calculation.UpdateCalculationTransaction(TransactionDocument, 0, ExchangeRate);
   410→            }
   411→        }

   656→        public void Recalculate()
   657→        {
   658→            if (Orders.Count > 0)
   659→            {
   660→                var orderGroup = Orders.GroupBy(x => x.AccountTransactionTypeId);
   661→                foreach (var orders in orderGroup)
   662→                {
   663→                    var o = orders;
   664→                    var transaction = TransactionDocument.AccountTransactions.SingleOrDefault(x => x.AccountTransactionTypeId == o.Key);
   665→                    if (transaction != null)
   666→                    {
   667→                        var amount = o.Sum(x => GetTaxExcludedSum(x));
   668→                        transaction.UpdateAccounts(GetTicketAccounts());
   669→                        transaction.UpdateAmount(amount, ExchangeRate);
   670→                    }
   671→                }
   672→
   673→                var taxIds = GetTaxIds();
   674→                if (taxIds.Any())
   675→                {
   676→                    var plainSum = GetPlainSum();
   677→                    var preTaxServices = GetPreTaxServicesTotal();
   678→                    foreach (var taxId in taxIds)
   679→                    {
   680→                        var transaction = TransactionDocument.AccountTransactions.Single(x => x.AccountTransactionTypeId == taxId);
   681→                        transaction.UpdateAccounts(GetTicketAccounts());
   682→                        transaction.UpdateAmount(GetTaxTotal(taxId, preTaxServices, plainSum), ExchangeRate);
   683→                    }
   684→                }
   685→            }
   686→
   687→            RemainingAmount = GetRemainingAmount();
   688→            TotalAmount = GetSum();
   689→        }

   716→        public Order ExtractSelectedOrder(Order order)
   717→        {
   718→            if (order.SelectedQuantity > 0 && order.SelectedQuantity < order.Quantity)
   719→            {
   720→                return FixOrder(order);
   721→            }
   722→            return order;

   725→        private List<Order> FixSelectedOrders()
   726→        {
   727→            return Orders.Where(x => x.SelectedQuantity > 0 && x.SelectedQuantity < x.Quantity)
   728→                         .ToList()
   729→                         .Select(FixOrder)
   730→                         .Where(newItem => newItem != null)
   731→                         .ToList();
   732→        }
   733→
   734→        private Order FixOrder(Order order)
   735→        {
   736→            Debug.Assert(order.SelectedQuantity > 0);
   737→            Debug.Assert(Orders.Contains(order));
   738→            if (order.SelectedQuantity >= order.Quantity) return null;
   739→            var result = CloneOrder(order);
   740→            result.Id = 0;
   741→            result.Quantity = order.SelectedQuantity;
   742→            result.ResetSelectedQuantity();
   743→            order.Quantity -= order.SelectedQuantity;
   744→            order.ResetSelectedQuantity();
   745→            return result;
   746→        }
