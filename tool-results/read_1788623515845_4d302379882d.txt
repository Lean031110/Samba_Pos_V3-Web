    25→function authGet(path) {
    26→  return request.get(path).set('Authorization', 'Bearer ' + jwtToken);
    27→}
    28→function authPost(path, body) {
    29→  return request.post(path).set('Authorization', 'Bearer ' + jwtToken).send(body || {});
    30→}
    31→function authPatch(path, body) {
    32→  return request.patch(path).set('Authorization', 'Bearer ' + jwtToken).send(body || {});
    33→}

   102→async function teardownFixtures() {
   103→  // Robust cleanup: only delete test-created rows, never seed data.
   104→  // Strategy: collect all ticket IDs created during tests, delete their children,
   105→  // then the tickets themselves. Then delete test menu items + tables.
   106→  try {
   107→    // Collect ticket IDs from all our test vars (some may be null if a test failed)
   108→    const ticketIds = [testTicketId].filter(Boolean);
   109→    // Also clean up any tickets with Note containing 'Sprint 5' or test markers
   110→    const extraTicketIds = await db('Tickets')
   111→      .whereLike('Note', '%Sprint 5%')
   112→      .orWhereLike('Note', '%Test%')
   113→      .orWhereLike('TicketStates', '%Refunded%')
   114→      .orWhereLike('TicketStates', '%Merged%')
   115→      .pluck('Id');
   116→    const allTicketIds = [...new Set([...ticketIds, ...extraTicketIds])];
   117→    if (allTicketIds.length > 0) {
   118→      await db('Orders').whereIn('TicketId', allTicketIds).del();
   119→      await db('Payments').whereIn('TicketId', allTicketIds).del();
   120→      await db('Calculations').whereIn('TicketId', allTicketIds).del();
   121→      await db('TicketEntities').whereIn('TicketId', allTicketIds).del();
   122→      await db('PaidItems').whereIn('TicketId', allTicketIds).del();
   123→      await db('Tickets').whereIn('Id', allTicketIds).del();
   124→    }
   125→    // Test menu item
   126→    if (testMenuItemId) {
   127→      const portionIds = await db('MenuItemPortions').where({ MenuItemId: testMenuItemId }).pluck('Id');
   128→      if (portionIds.length > 0) {
   129→        await db('MenuItemPrices').whereIn('MenuItemPortionId', portionIds).del();
   130→      }
   131→      await db('MenuItemPortions').where({ MenuItemId: testMenuItemId }).del();
   132→      await db('MenuItems').where({ Id: testMenuItemId }).del();
   133→    }
   134→    // Test table
   135→    if (testTableId) {
   136→      await db('EntityStateValues').where({ EntityId: testTableId }).del();
   137→      await db('Entities').where({ Id: testTableId }).del();
   138→    }
   139→    // Test tax templates + txn types (names start with 'Test ')
   140→    const testTaxTemplateIds = await db('TaxTemplates').whereLike('Name', 'Test %').pluck('Id');
   141→    if (testTaxTemplateIds.length > 0) {
   142→      await db('TaxTemplateMaps').whereIn('TaxTemplateId', testTaxTemplateIds).del();
   143→      await db('TaxTemplates').whereIn('Id', testTaxTemplateIds).del();
   144→    }
   145→    const testTxnTypeIds = await db('AccountTransactionTypes').whereLike('Name', 'Test %').pluck('Id');
   146→    if (testTxnTypeIds.length > 0) {
   147→      await db('AccountTransactionTypes').whereIn('Id', testTxnTypeIds).del();
   148→    }
   149→  } catch (err) {
   150→    console.error('[teardown] Error (continuing):', err.message);
   151→  }
   152→  await db.destroy();
   153→}

   156→// ANSI colors + counter
   157→// =====================================================================
   158→const C = {
   159→  reset: '\x1b[0m', bold: '\x1b[1m',
   160→  red: '\x1b[31m', green: '\x1b[32m', yellow: '\x1b[33m',
   161→  cyan: '\x1b[36m', gray: '\x1b[90m',
   162→};
   163→
   164→let passedCount = 0;
   165→let failedCount = 0;
   166→
   167→function header(title) {
   168→  console.log('\n' + C.bold + C.cyan + '═'.repeat(70) + C.reset);
   169→  console.log(C.bold + C.cyan + '  ' + title + C.reset);
   170→  console.log(C.bold + C.cyan + '═'.repeat(70) + C.reset);
   171→}
   172→
   173→function ok(msg) { console.log(C.green + '  ✓ ' + C.reset + msg); passedCount++; }
   174→function fail(msg) { console.log(C.red + '  ✗ ' + C.reset + msg); failedCount++; }
   175→
   176→// Override test() to wrap with try/catch and counting
   177→const originalTest = test;
   178→function countedTest(name, fn) {
   179→  return originalTest(name, async (t) => {
   180→    try {
   181→      await fn(t);
   182→    } catch (err) {
   183→      fail(name + ' — ' + err.message);
   184→      throw err;
   185→    }
   186→  });
   187→}
