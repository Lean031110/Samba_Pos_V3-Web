   139→   139→        public TicketCommitResult CloseTicket(Ticket ticket)
   140→   140→        {
   141→   141→            var result = _ticketDao.CheckConcurrency(ticket);
   142→   142→            Debug.Assert(ticket != null);
   143→   143→            var changed = !string.IsNullOrEmpty(result.ErrorMessage);
   144→   144→            var canSumbitTicket = !changed && ticket.CanSubmit;
   145→   145→
   146→   146→            if (canSumbitTicket)
   147→   147→            {
   148→   148→                RecalculateTicket(ticket);
   149→   149→                _applicationState.NotifyEvent(RuleEventNames.BeforeTicketClosing, new { Ticket = ticket, TicketId = ticket.Id, ticket.RemainingAmount, ticket.TotalAmount });
   150→   150→                if (ticket.Orders.Count > 0)
   151→   151→                {
   152→   152→                    var ticketType = _cacheService.GetTicketTypeById(ticket.TicketTypeId);
   153→   153→
   154→   154→                    if (ticket.Orders.Any(x => x.OrderNumber == 0))
   155→   155→                    {
   156→   156→                        var number = _settingService.GetNextNumber(ticketType.OrderNumerator.Id);
   157→   157→                        ticket.MergeOrdersAndUpdateOrderNumbers(number);
   158→   158→                    }
   159→   159→
   160→   160→                    if (ticket.Id == 0)
   161→   161→                    {
   162→   162→                        UpdateTicketNumber(ticket, ticketType.TicketNumerator);
   163→   163→                        _ticketDao.Save(ticket);
   164→   164→                    }
   165→   165→
   166→   166→                    Debug.Assert(!string.IsNullOrEmpty(ticket.TicketNumber));
   167→   167→                    Debug.Assert(ticket.Id > 0);
   168→   168→                    _applicationState.NotifyEvent(RuleEventNames.TicketClosing, new { Ticket = ticket, TicketId = ticket.Id, ticket.RemainingAmount, ticket.TotalAmount });
   169→   169→                    ticket.LockTicket();
   170→   170→                }
   171→   171→
   172→   172→                ticket.RemoveZeroAmountAccountTransactions();
   173→   173→
   174→   174→                if (ticket.Id > 0)// eğer adisyonda satır yoksa ID burada 0 olmalı.
   175→   175→                    _ticketDao.Save(ticket);
   176→   176→
   177→   177→                Debug.Assert(ticket.Orders.Count(x => x.OrderNumber == 0) == 0);
   178→   178→            }
   179→   179→
   180→   180→            if (ticket.Id > 0)
   181→   181→            {
   182→   182→                foreach (var ticketEntity in ticket.TicketEntities)
   183→   183→                {
   184→   184→                    var entityType = _cacheService.GetEntityTypeById(ticketEntity.EntityTypeId);
   185→   185→                    _applicationState.NotifyEvent(RuleEventNames.EntityUpdated, new
   186→   186→                    {
   187→   187→                        EntityTypeId = ticketEntity.EntityTypeId,
   188→   188→                        EntityId = ticketEntity.EntityId,
   189→   189→                        EntityTypeName = entityType.Name,
   190→   190→                        OpenTicketCount = _ticketServiceBase.GetOpenTicketIds(ticketEntity.EntityId).Count()
   191→   191→                    });
   192→   192→                }
   193→   193→            }
   194→   194→
   195→   195→            result.TicketId = ticket.Id;
   196→   196→            return result;
   197→   197→        }

   214→   214→                    RemainingAmount = ticket.GetRemainingAmount()

   254→   254→        public TicketCommitResult MergeTickets(IEnumerable<int> ticketIds)
   255→   255→        {
   256→   256→            var ticketList = ticketIds.Select(OpenTicket).ToList();
   257→   257→
   258→   258→            if (ticketList.Any(x => x.Calculations.Any()))
   259→   259→                return new TicketCommitResult { ErrorMessage = string.Format("Can't merge tickets\r{0}", "contains calculations") };
   260→   260→
   261→   261→            var entitiesUnMatches = ticketList.SelectMany(x => x.TicketEntities).GroupBy(x => x.EntityTypeId).Any(x => x.Select(y => y.EntityId).Distinct().Count() > 1);
   262→   262→            if (entitiesUnMatches) return new TicketCommitResult { ErrorMessage = string.Format("Can't merge tickets\r{0}", "Entities doesn't match") };
   263→   263→
   264→   264→            var clonedOrders = ticketList.SelectMany(x => x.Orders).Select(ObjectCloner.Clone).ToList();
   265→   265→            var clonedPayments = ticketList.SelectMany(x => x.Payments).Select(ObjectCloner.Clone2).ToList();
   266→   266→            var clonedChangePayments = ticketList.SelectMany(x => x.ChangePayments).Select(ObjectCloner.Clone2).ToList();
   267→   267→            var clonedTags = ticketList.SelectMany(x => x.GetTicketTagValues()).Select(ObjectCloner.Clone).ToList();
   268→   268→            var clonedEntites = ticketList.SelectMany(x => x.TicketEntities).Select(ObjectCloner.Clone).ToList();
   269→   269→            var clonedLogs = ticketList.SelectMany(x => x.GetTicketLogValues()).Select(ObjectCloner.Clone).ToList();
   270→   270→            var ticketNumbers = string.Join(",", ticketList.Select(x => x.TicketNumber));
   271→   271→
   272→   272→            ticketList.ForEach(x => x.RemoveData());
   273→   273→            ticketList.ForEach(x => CloseTicket(x));
   274→   274→
   275→   275→            var ticket = CreateTicket();
   276→   276→
   277→   277→            clonedOrders.ForEach(ticket.Orders.Add);
   278→   278→
   279→   279→            foreach (var cp in clonedPayments)
   280→   280→            {
   281→   281→                var account = _accountService.GetAccountById(cp.AccountTransaction.TargetTransactionValue.AccountId);
   282→   282→                ticket.AddPayment(_cacheService.GetPaymentTypeById(cp.PaymentTypeId), account, cp.Amount, GetExchangeRate(account), cp.UserId);
   283→   283→            }
   284→   284→
   285→   285→            foreach (var cp in clonedChangePayments)
   286→   286→            {
   287→   287→                var account = _accountService.GetAccountById(cp.AccountTransaction.TargetTransactionValue.AccountId);
   288→   288→                ticket.AddChangePayment(_cacheService.GetChangePaymentTypeById(cp.ChangePaymentTypeId), account, cp.Amount, GetExchangeRate(account), cp.UserId);
   289→   289→            }
   290→   290→
   291→   291→            clonedEntites.ForEach(x => ticket.UpdateEntity(x.EntityTypeId, x.EntityId, x.EntityName, x.AccountTypeId, x.AccountId, x.EntityCustomData));
   292→   292→            clonedTags.ForEach(x => ticket.SetTagValue(x.TagName, x.TagValue));
   293→   293→
   294→   294→            ticket.SetLogs(clonedLogs);
   295→   295→
   296→   296→            RefreshAccountTransactions(ticket);
   297→   297→
   298→   298→            _applicationState.NotifyEvent(RuleEventNames.TicketsMerged, new { Ticket = ticket, TicketNumbers = ticketNumbers });
   299→   299→            return CloseTicket(ticket);
   300→   300→        }
